[SCAN_1__BASELINE_XG_NPXG]
  GOAL : "填 Home_npxG_For_Avg / Home_xGA_Avg / Away_... 以及 Last5 序列"
  ACTION:
    STEP_A: "开 Understat league表 → 找到两队行 → 抄 xG / xGA / npxG / MP"
    SUCCESS_FILL:
      Home_npxG_For_Avg = npxG / MP
      Home_xGA_Avg     = xGA / MP
      SRC_TAG = "UNDERSTAT"
    FAIL:
      IF only SofaScore single-match xG available → fill RAW_XG_DISPLAY_AVG
          SRC_TAG = "SOFASCORE_DISPLAY [口径未公开·降级]"
          FLAG = [XG_MODEL_MISMATCH]
      ELSE → Home_npxG_For_Avg = UNKNOWN → FLAG = [NO_NPXG_HIST] → 走保守λ（同V16.2 §3.1-alt）

[SCAN_2__XT_THREAT_ZSCORE]
  GOAL : "回答：哪边创造更高质量机会？（序数）"
  METHOD: "
    实际可搜到的不是xT值·而是一组可观察代理：
    - Understat team页逐场Shot列表 → 手工目测 shot zone分布（box vs outbox）
    - SofaScore match页 Heatmap → 看进攻集中侧（左/右/中路）

  "
  OUTPUT_FORMAT: "
    xT_note = 'Home 威胁集中在右肋区(大量SofaScore热度) vs Away宽区低效'

    Z_A_npxG_rel = (Away_npxG_For_Avg - league_avg_npxG) / league_std_npxG
    ← Z-Score 只用 npxG 算（league_avg / league_std 从 Understat 表底部/手工算）
    ← xT '值'本身不进λ
  "
  NOTES: "V15.1原文的'Z-Score对npxG和xT归一化'→修正为：Z-Score只对npxG做·xT降级为热力描述"
